\begin{frame}{Come riconoscere un volto mai visto prima?}
\begin{columns}[T, onlytextwidth]

%% ---------- colonna sinistra ----------
\begin{column}{0.56\textwidth}
    \textbf{Il limite dei classificatori tradizionali}\\[0.35em]
    {\small Un modello addestrato su $N$ identità non può generalizzare
    su identità \warn{mai viste}: le feature sono ancorate alle classi
    di training, non alla geometria del volto.}

    \medskip
    \textbf{La soluzione: Metric Learning}\\[0.25em]
    {\small Impariamo $f:\mathcal{I}\to\mathbb{R}^{512}$ tale che:
    \[
      d\!\left(f(x_i),f(x_j)\right)\;\ll\; d\!\left(f(x_i),f(x_k)\right)
    \]
    quando $x_i,x_j$ sono la \good{stessa persona}
    e $x_k$ è una \bad{persona diversa}.}

    \medskip
    \begin{itemize}
        \item Variazioni di posa, illuminazione, espressione
        \item Generalizzazione \emphc{zero-shot} su nuove identità
        \item Valutazione: \emphc{mAP@1 / @5 / @10} su identità unseen
    \end{itemize}
\end{column}

\begin{frame}{Il metodo: Triplet Loss con PK Sampling e Online Hard Mining}
\begin{columns}[T, onlytextwidth]

%% ---------- colonna sinistra ----------
\begin{column}{0.46\textwidth}
    \textbf{La Tripletta}\\[0.45em]
    \begin{tikzpicture}[font=\scriptsize, node distance=0.25cm]
        % Anchor
        \node[circle, draw=colPrimary, fill=colPrimary, text=white,
              minimum size=0.92cm, font=\small\bfseries] (A) {A};
        \node[below=0.05cm of A, colPrimary, font=\tiny] {Anchor};
        % Positive
        \node[circle, draw=colGreen, fill=colGreen, text=white,
              minimum size=0.92cm, font=\small\bfseries,
              above right=0.60cm and 1.20cm of A] (P) {P};
        \node[above=0.05cm of P, colGreen, font=\tiny] {Positive (stessa ID)};
        % Negative
        \node[circle, draw=colRed, fill=colRed, text=white,
              minimum size=0.92cm, font=\small\bfseries,
              below right=1.05cm and 2.20cm of A] (N) {N};
        \node[below=0.05cm of N, colRed, font=\tiny] {Negative (diversa ID)};
        % Frecce
        \draw[Stealth-, colGreen, thick] (A.north east) -- (P.south west)
              node[midway, above, sloped, font=\tiny] {\good{vicini}};
        \draw[-Stealth, colRed, thick] (A.south east) -- (N.west)
              node[midway, below, sloped, font=\tiny] {\bad{lontani}};
    \end{tikzpicture}

    \medskip
    \textbf{PK Sampling — batch efficaci:}
    \begin{itemize}
        \item $P = 32$ identità $\times$ $K = 4$ immagini/ID
        \item Batch effettivo: $128$ immagini bilanciate
        \item Garantisce mining \emphc{ricco e stabile}
    \end{itemize}
\end{column}

%% ---------- colonna destra ----------
\begin{column}{0.50\textwidth}
    \textbf{Online Mining batch-wise}\\[0.35em]
    \small
    \begin{enumerate}
        \item Estrai embedding per il batch (128 immagini)
        \item Calcola matrice distanze $D\in\mathbb{R}^{128\times128}$ (cosine)
        \item Per ogni anchor seleziona:
        \begin{itemize}
            \item \warn{Hard positive}: positivo \emph{più lontano}
            \item \warn{Hard negative}: negativo \emph{più vicino}
        \end{itemize}
        \item Calcola la loss solo sui triplet difficili
    \end{enumerate}

    \medskip
    \begin{block}{\small Perché hard mining?}
        \small I triplet \emph{easy} producono gradiente $\approx 0$:\\
        il modello non impara nulla da essi.\\
        Hard mining seleziona i casi più \emphc{informativi}.
    \end{block}

    \medskip
    {\small Distanza usata:
    $d(u,v) = 1 - \langle u,v\rangle$
    su embedding L2-normalizzati.}
\end{column}

\end{columns}
\end{frame}

% SLIDE 6 — SOFT vs HINGE

\begin{frame}{La formulazione della loss conta più del margine: soft batte hinge}
\begin{columns}[T, onlytextwidth]

%% ---------- colonna sinistra ----------
\begin{column}{0.47\textwidth}
    \begin{alertblock}{\warn{[!]} Hinge + Semi-Hard stabile ma lenta}
        \vspace*{-0.3cm}
        \[
          \mathcal{L}_{\mathrm{hinge}} =
          \max\!\bigl(0,\; d(x_a, x_p) - d(x_a, x_n) + m\bigr)
        \]
        {\small Evita il collasso dovuto al label noise, ma la convergenza con margine rigido risulta più lenta e meno efficiente rispetto alla controparte soft.}
    \end{alertblock}

    \medskip
    \includegraphics[width=\linewidth, height=3.5cm,
                     keepaspectratio]{hinge_semihard.png}\\
    {\tiny\centering\color{colText!55!white}
        Curve hinge + semi-hard mining: solido\par}
\end{column}

%% ---------- colonna destra ----------
\begin{column}{0.47\textwidth}
    \begin{exampleblock}{SoftPlus + Hard Mining robusta al rumore}
        \[
          \mathcal{L}_{\mathrm{soft}} =
          \ln\!\Bigl(1 + e^{d(x_a, x_p) - d(x_a, x_n)}\Bigr)
        \]
        {\small Nessun margine fisso: penalizzazione continua e
        proporzionale. Tollerante al noise $\to$ convergenza
        stabile.}
    \end{exampleblock}

    \medskip
    \includegraphics[width=\linewidth, height=3.5cm,
                     keepaspectratio]{soft.png}\\
    {\tiny\centering\color{colText!55!white}
        Curve softmargin + hard mining: convergenza pulita\par}
\end{column}

\begin{frame}{Perché non possiamo ottimizzare Hinge con l'Hard Mining?}
\vspace*{-0.35cm}
\begin{columns}[T, onlytextwidth]

%% ---------- colonna sinistra ----------
\begin{column}{0.48\textwidth}
    \begin{alertblock}{{\color{colAccent}\textbf{[Instabile]}} Hinge + Hard Mining}
        \scriptsize
        La combinazione impedisce una convergenza stabile, degradando lo spazio latente (\textbf{mAP@1 $\approx$ 25\%}).
    \end{alertblock}
    
    \smallskip
    {\centering
    \includegraphics[width=0.95\linewidth, height=2.1cm, keepaspectratio]{hinge.png}\\
    \vskip -0.15cm
    {\tiny\color{colText!55!white}
        Hinge + Hard Mining: gradienti instabili e forti oscillazioni\par}
    \par}
        
    \smallskip
    {\scriptsize
    \begin{itemize}
        \setlength{\itemsep}{2pt}
        \setlength{\parskip}{0pt}
        \item \textbf{Sensibilità agli Outlier}: L'hard mining seleziona negativi atipici o etichettati erroneamente (rumore/outliers) trattandoli come falsi negativi estremi.
        \item \textbf{Margine Rigido}: Hinge impone una penale e un gradiente costante su questi errori, corrompendo lo spazio.
    \end{itemize}
    }
\end{column}

%% ---------- colonna destra ----------
\begin{column}{0.48\textwidth}
    \begin{exampleblock}{{\color{colAccent}\textbf{[Stabile]}} Soft Margin + Hard Mining}
        \scriptsize
        Convergenza rapida e stabile (\textbf{mAP@1 $\approx$ 87\%}).
    \end{exampleblock}
    
    \smallskip
    {\centering
    \includegraphics[width=0.95\linewidth, height=2.1cm, keepaspectratio]{soft.png}\\
    \vskip -0.15cm
    {\tiny\color{colText!55!white}
        Soft Margin + Hard Mining: convergenza robusta\par}
    \par}
        
    \smallskip
    {\scriptsize
    \begin{itemize}
        \setlength{\itemsep}{2pt}
        \setlength{\parskip}{0pt}
        \item \textbf{Gradiente Fluido}: La loss `softplus` attenua in modo asintotico l'influenza dei campioni troppo difficili.
        \item \textbf{Robustezza Intrinseca}: Il modello tollera le anomalie senza indebolire la durezza del mining.
    \end{itemize}
    }
\end{column}

\end{columns}
\end{frame}

\begin{frame}{Conclusioni: soft mining + PK batch = +24 p.p. zero-shot}
\vspace*{-0.35cm}
\begin{columns}[T, onlytextwidth]

%% ---------- colonna sinistra ----------
\begin{column}{0.54\textwidth}
    \begin{exampleblock}{Messaggio principale}
        \small
        Il Triplet con \emphc{soft loss} e \emphc{PK(32,4)} porta il mAP@1
        \textbf{dal 63\% all'87\%}, superando anche ArcFace
        (+3.1 p.p.)\\[0.3em]
        La chiave: la formulazione \emph{soft} è più importante
        del valore del margine, specialmente con dataset rumorosi.
    \end{exampleblock}

    \medskip
    \textbf{3 Takeaway concreti:}
    \begin{enumerate}
        \item \good{Soft $>$ Hinge} con label noise elevato:\\
              il margine esplicito è instabile con CASIA
        \item \good{PK(32,4) + hard mining} è il compromesso ottimale:\\
              batch più piccoli portano a overfitting
        \item \good{Hard mining è necessario}:\\
              mining easy/semi crea un gap di oltre 17 p.p.
    \end{enumerate}
\end{column}

%% ---------- colonna destra ----------
\begin{column}{0.42\textwidth}
    \begin{alertblock}{\warn{[!]} Limiti riconosciuti}
        \small
        \begin{itemize}
            \item \textbf{Mislabeling CASIA} (7.7\%–13\%):\\
                  avvelena il gradiente con hard mining
            \item \textbf{Assenza di Face Alignment}:\\
                  no MTCNN/landmark cropping\\
                  $\to$ variabilità non ridotta preventivamente
        \end{itemize}
    \end{alertblock}

    \medskip
    \textbf{Lavoro futuro:}
    \begin{itemize}
        \item Noise detection semi-supervisionata
        \item Face alignment preprocessing (MTCNN)
    \end{itemize}
\end{column}

\begin{frame}{Differenze Teoriche: Triplet Loss vs ArcFace}
\begin{columns}[T, onlytextwidth]

%% ---------- colonna sinistra ----------
\begin{column}{0.48\textwidth}
    \begin{block}{\small Triplet Loss: Confronto Relativo (Locale)}
        \scriptsize
        Ottimizza le distanze relative all'interno di mini-batch locali (tuple di 3 campioni).
    \end{block}

    \smallskip
    \begin{itemize}
        \setlength{\itemsep}{2pt}
        \setlength{\parskip}{0pt}
        \scriptsize
        \item \textbf{Formulazione}: Spinge $d(x_a, x_p) \to 0$ e allontana $d(x_a, x_n)$ oltre un margine $m$:
        \[
          \mathcal{L} = \max\!\bigl(0, d(x_a,x_p) - d(x_a,x_n) + m\bigr)
        \]
        \item \textbf{Spazio Latente}: Struttura locale flessibile; le classi non hanno un centro fisso e possono disperdersi.
        \item \textbf{Limiti}:
            \begin{itemize}
                \scriptsize
                \item Complessità combinatoria delle triplette $O(N^3)$.
                \item Richiede algoritmi di \emph{mining} online complessi.
                \item Sensibile al label noise locale.
            \end{itemize}
    \end{itemize}
\end{column}

%% ---------- colonna destra ----------
\begin{column}{0.48\textwidth}
    \begin{block}{\small ArcFace: Margine Angolare (Globale)}
        \scriptsize
        Approccio basato su classificazione softmax con margine additivo applicato geodeticamente sull'ipersfera unitaria.
    \end{block}

    \smallskip
    \begin{itemize}
        \setlength{\itemsep}{2pt}
        \setlength{\parskip}{0pt}
        \scriptsize
        \item \textbf{Formulazione}: Modifica la logit inserendo un margine angolare $m$ rispetto al peso-prototipo $W_j$:
        \[
          \mathcal{L} = -\log \frac{e^{s \cos(\theta_{y_i} + m)}}{e^{s \cos(\theta_{y_i} + m)} + \sum_{j \neq y_i} e^{s \cos \theta_j}}
        \]
        \item \textbf{Spazio Latente}: Gli embedding sono proiettati su $S^{d-1}$. Le classi formano cluster iper-compatti attorno a prototipi globali $W_j$.
        \item \textbf{Vantaggi e Limiti}:
            \begin{itemize}
                \scriptsize
                \item Nessun mining richiesto; training stabile.
                \item Complessità lineare con le identità $O(N \cdot C)$.
                \item Richiede molta memoria GPU per dataset con milioni di classi ($C$ elevato).
            \end{itemize}
    \end{itemize}
\end{column}

\end{columns}
\end{frame}
